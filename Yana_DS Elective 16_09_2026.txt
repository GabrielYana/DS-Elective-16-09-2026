DROP TABLE IF EXISTS EagleCharacteristics;
DROP TABLE IF EXISTS EagleSpecies;
GO

CREATE TABLE EagleSpecies (
    eagle_id int PRIMARY KEY,
    eagle_name varchar(40) NOT NULL UNIQUE,
    scientific_name varchar(80) NULL,
    region varchar(40) NOT NULL,
    habitat varchar(40) NOT NULL,
    conservation_status varchar(30) NOT NULL
);
GO

CREATE TABLE EagleCharacteristics (
    characteristic_id int IDENTITY(1,1) PRIMARY KEY,
    eagle_id int NOT NULL UNIQUE,
    average_wingspan_cm decimal(6,2) NOT NULL,
    average_weight_kg decimal(5,2) NOT NULL,
    average_lifespan_years int NOT NULL,
    primary_diet varchar(40) NOT NULL,
    FOREIGN KEY (eagle_id) REFERENCES EagleSpecies(eagle_id)
);
GO

-- 1: Which eagles have the largest average wingspans?
SELECT TOP 5
    e.eagle_name AS [Eagle],
    e.region AS [Region],
    c.average_wingspan_cm AS [Average Wingspan CM]
FROM EagleSpecies AS e
INNER JOIN EagleCharacteristics AS c
    ON e.eagle_id = c.eagle_id
ORDER BY c.average_wingspan_cm DESC;

-- 2: Which eagles live in forests or rainforests?
SELECT
    e.eagle_name AS [Eagle],
    e.region AS [Region],
    e.habitat AS [Habitat],
    e.conservation_status AS [Conservation Status]
FROM EagleSpecies AS e
WHERE e.habitat IN ('Forests', 'Rainforests', 'Tropical forests')
ORDER BY e.eagle_name;

-- 3: Which eagle species are threatened?
SELECT
    e.eagle_name AS [Eagle],
    e.region AS [Region],
    e.conservation_status AS [Conservation Status],
    c.primary_diet AS [Primary Diet]
FROM EagleSpecies AS e
INNER JOIN EagleCharacteristics AS c
    ON e.eagle_id = c.eagle_id
WHERE e.conservation_status IN ('Critically Endangered', 'Vulnerable', 'Near Threatened')
ORDER BY e.conservation_status, e.eagle_name;

-- Function
-- 4: What are the average wingspan, average weight, and longest lifespan?
SELECT
    AVG(c.average_wingspan_cm) AS [Average Wingspan CM],
    AVG(c.average_weight_kg) AS [Average Weight KG],
    MAX(c.average_lifespan_years) AS [Longest Lifespan Years]
FROM EagleCharacteristics AS c;

-- 5: How many eagles are recorded in each conservation-status category?
SELECT
    e.conservation_status AS [Conservation Status],
    COUNT(*) AS [Number of Eagles]
FROM EagleSpecies AS e
GROUP BY e.conservation_status
ORDER BY COUNT(*) DESC, e.conservation_status;

-- 6: Which eagles have a wingspan above the overall average?
SELECT
    UPPER(e.eagle_name) AS [Eagle Name],
    c.average_wingspan_cm AS [Wingspan CM],
    CAST(c.average_wingspan_cm -
        (SELECT AVG(average_wingspan_cm) FROM EagleCharacteristics)
        AS decimal(6,2)) AS [CM Above Average]
FROM EagleSpecies AS e
INNER JOIN EagleCharacteristics AS c
    ON e.eagle_id = c.eagle_id
WHERE c.average_wingspan_cm >
      (SELECT AVG(average_wingspan_cm) FROM EagleCharacteristics)
ORDER BY c.average_wingspan_cm DESC;
